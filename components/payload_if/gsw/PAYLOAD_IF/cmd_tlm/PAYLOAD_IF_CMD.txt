# PayOBC command (ICD section 5.2 layout) on APID 0x010: a telecommand without a
# secondary header (ICD RevB D6). PAYLOAD_IF forwards it unchanged over the payload
# link; with default values this is the ICD RevB A1 frame test vector packet.
COMMAND PAYLOAD_IF PAYOBC_CMD BIG_ENDIAN "Send a command to the PayOBC through PAYLOAD_IF"
  APPEND_PARAMETER CCSDS_STREAMID      16 UINT 0 65535 0x1010 "CCSDS Packet Identification"
  APPEND_PARAMETER CCSDS_SEQUENCE      16 UINT 0 65535 0xC000 "CCSDS Packet Sequence Control"
  APPEND_PARAMETER CCSDS_LENGTH        16 UINT 0 65535 6      "CCSDS Packet Data Length"
  APPEND_PARAMETER OPCODE               8 UINT 0 255   0x20   "PayOBC opcode (0x20 = Run Experiment)"
    FORMAT_STRING "0x%02X"
  APPEND_PARAMETER COMMAND_COUNTER     16 UINT 0 65535 1      "Command counter; the PayOBC echoes it in its status telemetry"
  APPEND_PARAMETER ARGUMENT            32 UINT 0 4294967295 1 "Command argument"
