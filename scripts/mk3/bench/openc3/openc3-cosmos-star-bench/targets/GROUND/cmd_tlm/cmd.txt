# Handled by sdr-link on the laptop and never sent to the busOBC. Each one makes the Pi put a
# bad frame on the real RS-422 line, to exercise the busOBC receive checks.
COMMAND GROUND INJECT_BAD_CRC BIG_ENDIAN "Frame with a wrong CRC. FRAME_ERR_COUNT +1"
  APPEND_PARAMETER CCSDS_STREAMID 16 UINT MIN_UINT16 MAX_UINT16 0x18F0 "CCSDS Packet Identification"
  APPEND_PARAMETER CCSDS_SEQUENCE 16 UINT MIN_UINT16 MAX_UINT16 0xC000 "CCSDS Packet Sequence Control"
  APPEND_PARAMETER CCSDS_LENGTH   16 UINT MIN_UINT16 MAX_UINT16 1 "CCSDS Packet Data Length"
  APPEND_PARAMETER CCSDS_FC        8 UINT MIN_UINT8 MAX_UINT8 1 "CCSDS Command Function Code"
  APPEND_PARAMETER CCSDS_CHECKSUM  8 UINT MIN_UINT8 MAX_UINT8 0 "CCSDS Command Checksum"
