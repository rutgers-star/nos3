# Commands for the Pi, carried by payload_if's forward command (code 6). payload_if checks the
# inner packet, then sends it over RS-422 unchanged. The inner layouts are what the Pi software
# reads today (PiDaemon payload_app.c).

  APPEND_PARAMETER CCSDS_STREAMID  16 UINT 0 65535 0x1850 "payload_if command MID"
  APPEND_PARAMETER CCSDS_SEQUENCE  16 UINT 0 65535 0xC000 "Sequence control"
  APPEND_PARAMETER CCSDS_LENGTH    16 UINT 0 65535 12 "Length"
  APPEND_PARAMETER CCSDS_FC         8 UINT 0 255 6 "Function code (6 = forward to the Pi)"
  APPEND_PARAMETER CCSDS_CHECKSUM   8 UINT 0 255 0 "Checksum"
  APPEND_PARAMETER PAYOBC_STREAMID 16 UINT 0 65535 0x1010 "PayOBC packet ID"
  APPEND_PARAMETER PAYOBC_SEQUENCE 16 UINT 0 65535 0xC000 "PayOBC sequence control"
  APPEND_PARAMETER PAYOBC_LENGTH   16 UINT 0 65535 4 "PayOBC packet data length"
  APPEND_PARAMETER OPCODE 8 UINT 3 3 3 "PayOBC opcode"
    FORMAT_STRING "0x%02X"
  APPEND_PARAMETER MET_MS 32 UINT 0 4294967295 1000000 "New mission elapsed time, ms"
