# Packets from the Pi as they reach the ground: inside payload_if's wrapper (MID 0x0862),
# stamped with the busOBC receive time. Data layouts are what the Pi software sends today
# (PiDaemon payload_app.c, ICD section 5). They come from a stand-in payload application, so
# the attitude requests and bulk data are test patterns, and temperature and storage are
# placeholders. MET and state are real program state.
TELEMETRY PAYOBC HEARTBEAT BIG_ENDIAN "Pi heartbeat, 1 Hz (APID 0x011)"
  APPEND_ID_ITEM CCSDS_STREAMID  16 UINT 0x0862 "payload_if wrapper for packets from the Pi"
  APPEND_ITEM    CCSDS_SEQUENCE  16 UINT "Sequence control"
  APPEND_ITEM    CCSDS_LENGTH    16 UINT "Length"
  APPEND_ITEM    BUSOBC_SECONDS  32 UINT "busOBC receive time (seconds)"
  APPEND_ITEM    BUSOBC_SUBSECS  16 UINT "busOBC receive time (subseconds)"
  APPEND_ITEM    CCSDS_SPARE     32 UINT ""
  APPEND_ID_ITEM PAYOBC_STREAMID 16 UINT 0x0011 "PayOBC packet ID"
  APPEND_ITEM    PAYOBC_SEQ_FLAGS 2 UINT "PayOBC sequence flags"
  APPEND_ITEM    SEQUENCE_COUNT  14 UINT "PayOBC packet count; goes up by one for every packet from the Pi"
  APPEND_ITEM    PAYOBC_LENGTH   16 UINT "PayOBC packet data length"
  APPEND_ITEM MET_MS       64 UINT "Pi mission elapsed time"
    UNITS milliseconds ms
  APPEND_ITEM OP_STATE      8 UINT "Pi operating state"
    STATE IDLE               0x00 GREEN
    STATE EXPERIMENT_RUNNING 0x01 YELLOW
    STATE FAULT              0xFF RED
  APPEND_ITEM EXPERIMENT_ID 8 UINT "Active experiment"
  APPEND_ITEM CPU_TEMP     32 FLOAT "Placeholder from the stand-in payload application"
    FORMAT_STRING "%.3f"
    UNITS celsius C
  APPEND_ITEM STORAGE_MB   32 UINT "Placeholder from the stand-in payload application"
    UNITS megabytes MB
